# Exercise every surviving ModelingService function plus native GeometryScript.
# Run in an exclusively owned editor session with execute_python. The directory
# must not already exist: cleanup removes only this script's temporary assets.
import unreal
import traceback

svc = unreal.ModelingService
ROOT = "/Game/Developers/Vibe3DModelingExercise"
handles = []
created_directory = False
failed = []
called = set()


def check(name, *args):
    called.add(name)
    result = getattr(svc, name)(*args)
    if hasattr(result, "success"):
        assert result.success, "%s: %s" % (name, result.message)
    elif isinstance(result, bool):
        assert result, name
    elif isinstance(result, int):
        assert result >= 0, name
    return result


def mesh():
    h = check("create_mesh").handle
    handles.append(h)
    return h


try:
    assert not unreal.EditorAssetLibrary.does_directory_exist(ROOT), "temporary directory already exists; refusing to overwrite or delete it"
    h = mesh()
    dm = check("get_dynamic_mesh", h)
    assert dm is not None and h in check("list_meshes")
    unreal.GeometryScript_Primitives.append_box(dm, unreal.GeometryScriptPrimitiveOptions(), unreal.Transform(), 100, 100, 100)
    assert unreal.GeometryScript_MeshQueries.get_is_closed_mesh(dm)
    unreal.GeometryScript_MeshRepair.compact_mesh(dm)
    unreal.GeometryScript_UVs.auto_generate_x_atlas_mesh_u_vs(dm, 0, unreal.GeometryScriptXAtlasOptions())
    stats = check("get_uv_stats", h, 0, 256)
    assert stats.num_unset_triangles == 0

    # A native selection crosses into the service and is used for rigging.
    outputs = unreal.GeometryScript_MeshSelection.create_select_all_mesh_selection(dm)
    selection = next(x for x in outputs if isinstance(x, unreal.GeometryScriptMeshSelection))
    assert check("set_selection", h, "all", selection) > 0
    B = unreal.ModelingBoneDef
    check("create_bones", h, [B(name="root"), B(name="flap", parent_name="root")])
    check("bind_selection_to_bone", h, "all", "flap", 1.0)
    assert check("select_by_bone_weight", h, "weighted", "flap", 0.25) > 0
    check("inspect_bone_weights", h, "weighted")
    check("scale_bone_weights", h, "weighted", "flap", 0.5, "root")
    check("smooth_selection_bone_weights", h, "weighted", 1, 0.5)
    check("clear_selections", h)

    # Save the newly rigged fixture and its skeleton only in the owned directory.
    created_directory = True
    sk = ROOT + "/SK_ExerciseRig"
    check("save_mesh_to_skeletal_mesh", h, sk, "", True, False)
    assert unreal.EditorAssetLibrary.does_asset_exist(sk + "_Skeleton")
    check("set_asset_materials", sk, "/Engine/BasicShapes/BasicShapeMaterial", False)
    tex = ROOT + "/T_ExerciseNoise"
    check("create_noise_texture", tex, 64, 64, 4.0, 3, 0.5, 3, False)
    check("draw_on_texture", tex, "Line", [unreal.Vector2D(0.1, 0.1), unreal.Vector2D(0.9, 0.9)], unreal.LinearColor(1, 1, 1, 1), 3.0, 8.0, False)

    # Use separate handles for the two retained ID-preserving primitives and stamps.
    part = mesh()
    check("append_torus", part, unreal.Transform(), 30, 10, 16, 8, "Base", 1)
    profile = [unreal.Vector2D(0, 0), unreal.Vector2D(10, 0), unreal.Vector2D(10, 20), unreal.Vector2D(0, 20)]
    check("append_revolve_polygon", part, unreal.Transform(), profile, 0, 16, 360, 2)
    detail = mesh()
    check("append_mesh_along_polyline", detail, part, [unreal.Vector(0, 0, 0), unreal.Vector(50, 0, 0)], 25, unreal.Vector(0, 0, 1), 1)
    groove = mesh()
    groove_dm = check("get_dynamic_mesh", groove)
    unreal.GeometryScript_Primitives.append_box(groove_dm, unreal.GeometryScriptPrimitiveOptions(), unreal.Transform(), 100, 100, 100)
    check("cut_groove_along_polyline", groove, [unreal.Vector(-60, 0, 100), unreal.Vector(60, 0, 100)], 2, 1, unreal.Vector(0, 0, 1))
    assert check("release_mesh", detail)
    handles.remove(detail)

    # Invalid handle must fail without affecting other meshes.
    invalid = svc.append_torus(-1, unreal.Transform())
    assert not invalid.success
except Exception:
    failed.append(traceback.format_exc())
finally:
    for handle in handles:
        try:
            if not svc.release_mesh(handle):
                failed.append("could not release handle %s" % handle)
        except Exception:
            failed.append(traceback.format_exc())
    # Exercise release_all_meshes only when no other session handles remain.
    try:
        if not svc.list_meshes():
            svc.release_all_meshes()
            called.add("release_all_meshes")
        else:
            failed.append("other session handles exist; release_all_meshes deliberately skipped")
    except Exception:
        failed.append(traceback.format_exc())
    try:
        if created_directory and unreal.EditorAssetLibrary.does_directory_exist(ROOT):
            if not unreal.EditorAssetLibrary.delete_directory(ROOT):
                failed.append("could not remove temporary asset directory")
        if created_directory and unreal.EditorAssetLibrary.does_directory_exist(ROOT):
            failed.append("temporary asset directory remains")
    except Exception:
        failed.append(traceback.format_exc())

expected = {"create_mesh", "release_mesh", "release_all_meshes", "list_meshes", "get_dynamic_mesh", "set_selection", "clear_selections", "append_torus", "append_revolve_polygon", "save_mesh_to_skeletal_mesh", "set_asset_materials", "create_bones", "bind_selection_to_bone", "select_by_bone_weight", "scale_bone_weights", "smooth_selection_bone_weights", "inspect_bone_weights", "get_uv_stats", "create_noise_texture", "draw_on_texture", "append_mesh_along_polyline", "cut_groove_along_polyline"}
missing = sorted(expected - called)
if missing:
    failed.append("not called: " + ", ".join(missing))
for error in failed:
    print("FAIL:", error)
print("RESULT:", "PASS" if not failed else "FAIL")
